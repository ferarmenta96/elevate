// Motor de Puntuación de Leads IA — HORUS
import { OpenAI } from 'openai';
import { db } from '@horus/database';

export async function puntuarLeadEntrante(lead) {
  const contexto = await db.leads
    .buscarSimilares(lead, { limit: 50 });

  const puntuacion = await ai.analizar({
    lead,
    datosHistoricos: contexto,
    modelo: 'horus-lead-v3'
  });

  if (puntuacion.probabilidad > 0.82) {
    await crm.asignarACierre(lead);
    await notificaciones.urgente(lead);
  }

  return { puntuacion, accion: puntuacion.siguientePaso };
}
